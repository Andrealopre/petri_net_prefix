174    def unfolding_algorithm(self):
175        """
176        La funzione che genera l'unfolding della rete di petri, ovvero l'aloritmo 4.2 del documento
177        Esparza, Romer, Vogler. Trova l'unfolding iniziale, corrispondente alla marcatura Iniziale, e le PE iniziali.
178        Infine ritorna l'unfolding della rete.
179
180        Attributes:
181
182            unf (set()): insieme che rappresenta l'unfolding della rete
183
184            pe (set()): insieme delle possible extensions
185
186            ev (Event): evento preso dalle possible extensions
187
188        Returns:
189
190            unf (set()): ritorna l'unfolding della rete di Petri
191        """
192
193        # Aggiungo all'unfolding la marcatura iniziale
194        unf = self.initialMarking()
195
196        # Trovo le estensioni iniziali, usando l'unfolding trovato in precedenza
197        pe = self.initial_extensions(unf)
198
199
200        """
201        Il ciclo while funziona solamente con reti senza cicli. Per poter utilizzare questo algoritmo con 
202        reti non finite, bisogna impostare un range finito.
203        Nel ciclo viene preso il primo evento disponibile presente nell'insieme di possible extensions, tale evento
204        viene aggiunto all'unfolding e all'insieme di eventi. Successivamente, ogni condizione presente nelle output
205        conditions dell'evento scelto viene aggiunta all'unfolding e all'insieme di condizioni. Viene aggiornata
206        la relazione di causalità post (evento -> condizione) e, per ogni condizione presente nelle input_conditions
207        dell'evento, viene aggiornata la relazione causale pre (condizione -> evento). Vengono poi aggiornati i 
208        conflitti dell'evento considerato. Infine vengono trovate le possible extension e vengono aggiunte a PE.
209        """
210        while pe:
211        #for x in range(30):
212
213            ev = pe.pop() #prende il primo evento disponibile dalla lista di estensioni
214
215
216            unf.add(ev) #aggiunge l'evento all'unfolding
217            self.events.add(ev)
218
219            # aggiungo ogni output_condition di ev ad unfolding
220            for cond in ev.output_conditions:
221                unf.add(cond)
222                self.causal_updater_post(ev, cond) #aggiorno relazione di causalità
223                self.conditions.add(cond)
224
225            # aggiorno causalità per ogni condizione di input dell'evento
226            for cond in ev.input_conditions:
227                self.causal_updater_pre(cond, ev)
228
229            # aggiorno i conflitti dell'evento considerato
230            self.update_conflict_relation(ev)
231
232            # chiamo la funzione che aggiorna le possible extensions
233            extensions = self.update_pot_ext({c for c in unf if isinstance(c, Condition)}, ev)
234
235            # estendo le possible extensions
236            pe |= extensions
237
238        return unf

286    def initial_extensions(self, unf):
287        """
288        La funzione che trova le possible extensions iniziali.
289        Incomincia prendendo tutte le condizioni dall'unfolding parziale passato come parametro. Successivamente
290        cicla ogni transizione presente nella rete di Petri iniziale e, per ogni transizione, ne trova il preset.
291        Vengono cercate le in_conditions, ovvero le condizioni di input da cui l'evento della transizione viene
292        generato. Si controlla se la lunghezza delle in_conditions sia uguale a quella dei preset_places, in caso
293        affermativo si crea il nuovo evento (new_event(t, in_conditions)) e viene aggiunto alle initial_extensions
294
295        Args:
296
297            unf (set()): unfolding parziale
298
299        Attributes:
300
301            initial_extensions (set()): insieme delle estensioni iniziali
302
303            initial_markings (set()): prende tutte le condizioni presenti nell'unfolding parziale
304
305            preset_places (set()): insieme del preset della transizione t
306
307            in_conditions (set()): insieme delle condizioni di input dell'evento etichettato da t. Si ottiene ciclando
308            sull'insieme initial_markings. Se la condizione c presa in cosiderazione ha il suo c.place all'interno
309            dei preset della transizione t, allora fa parte delle input_conditions
310
311            post_places (set()): insieme del postset della transizione t
312
313        Returns:
314
315            initial_extensions (set()): insieme delle possible extensions iniziali
316
317        """
318        initial_extensions = set()
319
320        # vengono prese tutte le condizioni in unf, possibilmente inutile, dato che si stanno trattando solo
321        # le initial_extensions, ma mai essere sicuri
322        initial_markings = {c for c in unf if isinstance(c, Condition)}
323
324        # ciclo for per ogni transizione t presente nella rete di Petri
325        for t in self.net.transitions.values():
326
327            # cerco i preset della transizione.
328            preset_places = self.find_preset(t)
329
330            # inserisco le condizioni di input il cui place è presente nei preset_places della transizione t
331            in_conditions = {c for c in initial_markings if c.place in preset_places}
332
333            if len(in_conditions) == len(preset_places):
334                new_event = Event(t, in_conditions) #creo il nuovo evento
335
336                # cerco i place successivi alla transizione e creo la condizione, inserendole nelle output_conditions
337                # dell'evento appena creato
338                post_places = {arc.dst for arc in self.net.arcs if arc.src == t}
339
340                for p in post_places:
341                    new_condition = Condition(p, new_event)
342                    new_event.output_conditions.add(new_condition)
343                
344                initial_extensions.add(new_event)       
345
346        return initial_extensions

348    def update_pot_ext(self, unf, e):
349        """
350        Funzione update_pot_ext ispirata dalla (quasi)medesima funzione presentata da Victor Khomenko nella sua tesi di
351        dottorato. Si occupa di filtrare le transizioni successive tramite le euristiche spiegate nella tesi. Dopo
352        cicla per ogni transizione, trova il preset della transizione considerata e trova tutte le condizioni c
353        presenti nell'unfolding che sono nel preset di t e che siano concorrenti con l'evento e iniziale.
354        Successivamente chiama la funzione cover.
355
356        Args:
357
358            unf (set()): unfolding parziale
359            e (Event): evento aggiunto all'unfolding dall'algoritmo principale
360        
361        Returns:
362
363            extensions (set()): possibili estensioni dell'evento e
364        """
365        extensions = set()
366
367        # Prendo la transizione etichettata dall'evento e
368        u = e.transition
369
370        post_places = self.find_post_places(u)  # u*
371        succ_transitions = self.find_succ_transitions(post_places)  # (u*)*
372
373        preset_places = self.find_preset(u)  # *u
374
375        difference = preset_places - post_places  # (*u \ u*)
376
377        post_difference = self.find_difference(difference)  # (*u \ u*)* postset della differenza
378
379        transitions = succ_transitions - post_difference  # (u*)* \ (*u \ u*)*
380
381        # cicla tutte le transizioni t nelle transizioni possibili
382        for t in transitions:
383            preset = self.find_preset(t)  # trova il preset di t
384
385            # ciclo che aggiunge a C le condizioni di unf, concorrenti all'evento e
386            C = {c for c in unf if isinstance(c, Condition) and c.place in preset and self.is_concurrent(c, e)}
387
388            extensions |= self.cover(C, t, dict())  # estende le estensioni grazie alla funzione di supporto cover
389
390        return extensions

392    def cover(self, C, t, preset):
393        """
394        Funzione che serve a completare il preset di un evento candidato.
395        Controlla le condizioni concorrenti e sceglie quelle che possono formare un preset
396        per una possibile estensione
397
398        Args:
399
400            C (set()): condizioni concorrenti all'evento e (trovate nella funzione extend_pot_ext())
401        
402            t: transizione candidata a diventare un nuovo evento
403
404            preset (dict()): un dizionario, rappresenta il preset incompleto di t
405        
406        Returns:
407
408            extensions (set()): la possibile estensione
409        """
410        pre_t = self.find_preset(t)
411        extensions = set()
412
413        # se il preset è completo, ovvero se è uguale alla cardinalità di *t entra 
414        # nell'if
415        if len(pre_t) == len(preset): # caso base
416
417            # trova le input_conditions in preset che si trovano nel 
418            # preset di t
419            in_conditions = {preset[p] for p in pre_t}
420
421            # se esiste già un evento uguale nell'unfolding, non si creanno duplicati
422            # ma si utilizza l'evento già inserito
423            existing = self.existing_event(t, in_conditions)
424            if existing:
425                return {existing} # ritorna evento già esistente
426
427            # altrimenti si crea un nuovo evento
428            new_event = Event(t, in_conditions)
429
430            # qui si trovano i posti successivi alla transizione t
431            # sono le output conditions
432            postset_places = {arc.dst for arc in self.net.arcs if arc.src == t}
433
434            for place in postset_places:
435                new_condition = Condition(place, new_event)
436                new_event.output_conditions.add(new_condition)
437
438            # si espandono le estensioni e si ritorna
439            extensions.add(new_event)
440            return extensions
441        else: # passo ricorsivo
442
443            # scelgo un place nel preset NON ancora presente nel preset che sto
444            # costruendo
445            diff = pre_t - set(preset.keys())
446
447            p = next(iter(diff))
448
449            # ciclo le condizioni in C e seleziono quelle che hanno il place uguale a quello scelto precedentemente
450            for d in [cond for cond in C if cond.place == p]:
451                # aggiungo a C_first (C') le condizioni in C concorrenti con la condizione d
452                C_first = {c for c in C if self.is_concurrent(c, d)}
453
454                # incremento il preset in costruzione
455                union_preset = dict(preset)
456                union_preset[p] = d
457
458                # estendo con chiamata ricorsiva
459                extensions |= self.cover(C_first, t, union_preset)
460
461            return extensions

477    def update_conflict_relation(self, event):
478        """
479        Funzione che aggiorna le relazioni di conflitto tra l'evento passato in input e tutti gli eventi presenti
480        nell'unfolding.
481        Controlla ogni evento presente in self.events (che contiene solo gli eventi già presenti nell'unfolding).
482        Propaga il conflitto agli eventi delle input conditions
483
484        Args:
485
486            event (Event): evento a cui si vogliono trovare i conflitti
487        """
488
489        for e in self.events:
490
491            # se "e" è l'evento stesso, continua
492            if e is event:
493                continue
494
495            # se la relazione di conflitto è già presente continua
496            if (event, e) in self.conflict_relation or (e, event) in self.conflict_relation:
497                continue
498
499            # se i preset di event e di "e" non hanno elementi in comune, aggiunge gli eventi nel conflitto
500            if not event.input_conditions.isdisjoint(e.input_conditions):
501                self.add_conflict(event, e)
502            
503            # Se gli eventi delle input conditions di event sono in conflitto con l'evento e, allora anche event
504            # sarà in conflitto con l'evento e
505            for c in event.input_conditions:
506                if c.event and ((c.event, e) in self.conflict_relation or (e, c.event) in self.conflict_relation):
507                    self.add_conflict(event, e)
508                    break

601    def is_causal(self, x, y):
602        """
603        Algoritmo BFS per controllare la causalità di due argomenti passati in input.
604        Se x e y sono condizioni, vengono rese eventi. Se sono condizioni iniziali, allora False.
605        Se sono lo stesso evento -> False
606        Se, ciclando per tutta la "linea" causale si trova l'evento y, allora ritorna True
607
608        Args:
609
610            x (Event or Condition)
611            y (Event or Condition)
612        
613        Returns:
614
615            True or False (Bool)
616        """
617
618        # Ottengo gli eventi di x e y se sono condizioni, altrimenti lascia invariato
619        x_ev = x.event if isinstance(x, Condition) else x
620        y_ev = y.event if isinstance(y, Condition) else y
621
622        # Se uno dei due eventi è nullo, allora non c'è causalità
623        if x_ev is None or y_ev is None:
624            return False
625        
626        # se gli eventi sono gli stessi PRIMA del ciclo, allora non sono causali
627        if x_ev is y_ev:
628            return False
629        
630        # inizializza un set di nodi visitati e una lista che parte dall'evento x
631        visited = set()
632        cone = [x_ev]
633
634        # estrae un nodo da cone, se node == y allora vuol dire che si è trovato y tra i successori
635        # e quindi c'è causalità
636        while cone:
637            node = cone.pop() # prende il primo elemento da cone e lo assegna a node
638
639            if node == y_ev: # nodo y trovato nei successori
640                return True
641            
642            # se il nodo è visitato continua
643            # aggiunge node a visited e crea il set di successori, che andranno ad estendere il cono
644            if node in visited:
645                continue
646            visited.add(node)
647
648            # condizioni prodotte dal nodo preso in considerazione
649            produced_conditions = {cond for (e, cond) in self.causal_relation_post if e is node}
650
651            # eventi successivi alle condizioni prodotte
652            successor_events = {ev2 for (cond2, ev2) in self.causal_relation_pre if cond2 in produced_conditions}
653
654            # estende il cono con i successor events
655            cone.extend(successor_events)
656        
657        # se y non è trovato ritorna falso
658        return False

680    def is_concurrent(self, n1, n2):
681        """
682        Funzione che controlla la concorrenza tra due eventi n1 e n2 passati input. La concorrenza è calcolata
683        "al volo", controllando la presenza di causalità e conflitti.
684
685        Args:
686
687            n1 (Event or Condition)
688            n2 (Evento or Condition)
689        """
690
691        # se n1 è n2 allora non sono concorrenti
692        if n1 is n2:
693            return False
694        
695        # ottengo gli eventi dei nodi, se sono condizioni
696        e1 = self.node_to_event(n1)
697        e2 = self.node_to_event(n2)
698
699        if e1 is not None and e2 is not None:
700            # se e1 ed e2 sono in relazione di causalità allora non sono concorrenti -> False
701            if self.is_causal(e1, e2) or self.is_causal(e2, e1):
702                return False
703            # se e1 ed e2 sono in conflitto -> False
704            if (e1, e2) in self.conflict_relation or (e2, e1) in self.conflict_relation:
705                return False
706            
707            # se gli eventi sono uguali -> false (se provo a fare e1 is e2 non funziona, devo capire perché)
708            if isinstance(n1, Condition) and isinstance(n2, Condition) and n1.event is n2.event:
709                return False
710
711            return True # se passo tutti i check, allora sono concorrenti
712        
713        # se n1 e n2 sono condizioni e sono le stesse allora torna False
714        # possibilmente ridondante, anche perché in teoria non si raggiunge mai questa parte del codice
715        # da valutare se eliminare completamente questo blocco di codice
716        if isinstance(n1, Condition) and isinstance(n2, Condition):
717            if n1 is n2:
718                return False
719            return True
720        
721        return True

827        sono chiare solamente gli eventi, dato che nel sito le condizioni non hanno il nome del place che

832            "places": [],
833            "transitions": [],
834            "arcs": {},
835            "marking": {},
836            "labels": {},
837            "actions": [],
838            "layout": {}
839        }
840
841        place_map = {}
842        trans_map = {}
843        place_idx = 0
844        trans_idx = 0
845
846        for n in unf:
847            # se il nodo è una condition riempio ogni campo del json con i dati di tale condition
848            # alla fine aumento index di 1
849            if isinstance(n, Condition):
850                pname = f"p{place_idx}"
851                place_map[n] = pname
852                data["places"].append(pname)
853                data["labels"][pname] = n.place.name
854                if n.event is None:
855                    data["marking"][pname] = 1
856                place_idx += 1
857            
858            # faccio la stessa cosa nel caso il nodo sia un evento
859            elif isinstance(n, Event):
860                tname = f"t{trans_idx}"
861                trans_map[n] = tname
862                data["transitions"].append(tname)
863                data["labels"][tname] = n.transition.name
864                data["actions"].append(n.transition.name)
865                trans_idx += 1
866
867        # questo blocco di codice fa in modo di non ripetere i nomi delle transizioni duplicate
868        seen = set()
869        actions_unique = []
870        for a in data["actions"]:
871            if a not in seen:
872                seen.add(a)
873                actions_unique.append(a)
874        data["actions"] = actions_unique
875        
876        # crea gli archi tra condizioni ed eventi (o il contrario) in base alla loro relazione di
877        # causalità
878        for (cond, ev) in self.causal_relation_pre:
879            src = place_map.get(cond)
880            dst = trans_map.get(ev)
881            if src and dst:
882                data["arcs"][f"{src},{dst}"] = 1
883
884        for (ev, cond) in self.causal_relation_post:
885            src = trans_map.get(ev)
886            dst = place_map.get(cond)
887            if src and dst:
888                data["arcs"][f"{src},{dst}"] = 1
889
890        # aggiunge tutti i nodi in unfolding
891        nodes_in_unf = set(n for n in unf)
892
893        # Costruzione del grafo che rappresenta l'ordine causale
894        # e calcolo dei gradi per ogni nodo
895        graph = defaultdict(list)
896        indegree = defaultdict(int)
897
898        # aumenta il livello di ogni nodo in base alla relazione causale
899        # se si è in un caso cond -> ev, allora alla condizione si "attacca" l'evento successivo
900        for (cond, ev) in self.causal_relation_pre:
901            if cond in nodes_in_unf and ev in nodes_in_unf:
902                graph[cond].append(ev)
903                indegree[ev] += 1
904                indegree.setdefault(cond, 0)
905        
906        # se si è in un caso ev -> cond, all'evento si attacca la condizione successiva
907        for (ev, cond) in self.causal_relation_post:
908            if ev in nodes_in_unf and cond in nodes_in_unf:
909                graph[ev].append(cond)
910                indegree[cond] += 1
911                indegree.setdefault(ev, 0)
912        
913        for n in nodes_in_unf:
914            indegree.setdefault(n, 0)
915
916        level = {} # distanza dalla marcatura iniziale
917
918        # Si inizializza la coda con i nodi senza predecessori, ovvero quelli al livello
919        # zero
920        queue = deque([n for n in unf if indegree[n] == 0])
921        for n in queue:
922            level[n] = 0
923
924        # visita topologica
925        # ogni nodo successore ottiene level = level predecessore + 1
926        while queue:
927            node = queue.popleft()
928            for succ in graph[node]:
929                level[succ] = max(level.get(succ, 0), level[node] + 1)
930                indegree[succ] -= 1
931
932                # quando arriva allo 0, tutti i predecessori sono stati elaborati
933                if indegree[succ] == 0:
934                    queue.append(succ)
935
936        # i nodi sono stratificati in base al loro livello
937        layers = defaultdict(list)
938        for n, l in level.items():
939
940            if isinstance(n, Condition) and n not in place_map:
941                continue
942            if isinstance(n, Event) and n not in trans_map:
943                continue
944            layers[l].append(n)
945
946        # distanze standard
947        spacing_x = 300
948        spacing_y = 100
949
950        # ad ogni nodo viene assegnata una posizione nel layout
951        # i nodi nello stesso livello sono verticali, i livelli successivi avanzano verticalmente
952        for l, nodes in sorted(layers.items()):
953            for i, n in enumerate(nodes):
954                x = spacing_x * l
955                y = spacing_y * i
956                if isinstance(n, Condition):
957                    name = place_map[n]
958                else:
959                    name = trans_map[n]
960                data["layout"][name] = {"x": x, "y": y}
961        
962        with open(filename, "w") as f:
963            json.dump(data, f, indent=4)
964        
965        print(f"File JSON dell'unfolding salvato in {filename}")

179        Infine ritorna l'unfolding della rete.

184

188

190

192        """
193
194        # Aggiungo all'unfolding la marcatura iniziale

196
197        # Trovo le estensioni iniziali, usando l'unfolding trovato in precedenza
198        pe = self.initial_extensions(unf)

205        viene aggiunto all'unfolding e all'insieme di eventi. Successivamente, ogni condizione presente nelle output
206        conditions dell'evento scelto viene aggiunta all'unfolding e all'insieme di condizioni. Viene aggiornata

213
214            ev = pe.pop() #prende il primo evento disponibile dalla lista di estensioni
215

218            self.events.add(ev)

221            for cond in ev.output_conditions:

310            dei preset della transizione t, allora fa parte delle input_conditions
311
312            post_places (set()): insieme del postset della transizione t
313
314        Returns:
315
316            initial_extensions (set()): insieme delle possible extensions iniziali
317
318        """
319        initial_extensions = set()
320
321        # vengono prese tutte le condizioni in unf, possibilmente inutile, dato che si stanno trattando solo
322        # le initial_extensions, ma mai essere sicuri
323        initial_markings = {c for c in unf if isinstance(c, Condition)}
324
325        # ciclo for per ogni transizione t presente nella rete di Petri
326        for t in self.net.transitions.values():
327
328            # cerco i preset della transizione.
329            preset_places = self.find_preset(t)
330
331            # inserisco le condizioni di input il cui place è presente nei preset_places della transizione t
332            in_conditions = {c for c in initial_markings if c.place in preset_places}
333
334            if len(in_conditions) == len(preset_places):
335                new_event = Event(t, in_conditions) #creo il nuovo evento
336
337                # cerco i place successivi alla transizione e creo la condizione, inserendole nelle output_conditions
338                # dell'evento appena creato
339                post_places = {arc.dst for arc in self.net.arcs if arc.src == t}
340
341                for p in post_places:
342                    new_condition = Condition(p, new_event)
343                    new_event.output_conditions.add(new_condition)
344                
345                initial_extensions.add(new_event)       
346
347        return initial_extensions

349    def update_pot_ext(self, unf, e):
350        """
351        Funzione update_pot_ext ispirata dalla (quasi)medesima funzione presentata da Victor Khomenko nella sua tesi di
352        dottorato. Si occupa di filtrare le transizioni successive tramite le euristiche spiegate nella tesi. Dopo
353        cicla per ogni transizione, trova il preset della transizione considerata e trova tutte le condizioni c
354        presenti nell'unfolding che sono nel preset di t e che siano concorrenti con l'evento e iniziale.
355        Successivamente chiama la funzione cover.
356
357        Args:
358
359            unf (set()): unfolding parziale
360            e (Event): evento aggiunto all'unfolding dall'algoritmo principale
361        
362        Returns:
363
364            extensions (set()): possibili estensioni dell'evento e
365        """
366        extensions = set()
367
368        # Prendo la transizione etichettata dall'evento e
369        u = e.transition
370
371        post_places = self.find_post_places(u)  # u*
372        succ_transitions = self.find_succ_transitions(post_places)  # (u*)*
373
374        preset_places = self.find_preset(u)  # *u
375
376        difference = preset_places - post_places  # (*u \ u*)
377
378        post_difference = self.find_difference(difference)  # (*u \ u*)* postset della differenza
379
380        transitions = succ_transitions - post_difference  # (u*)* \ (*u \ u*)*
381
382        # cicla tutte le transizioni t nelle transizioni possibili
383        for t in transitions:
384            preset = self.find_preset(t)  # trova il preset di t
385
386            # ciclo che aggiunge a C le condizioni di unf, concorrenti all'evento e
387            C = {c for c in unf if isinstance(c, Condition) and c.place in preset and self.is_concurrent(c, e)}
388
389            extensions |= self.cover(C, t, dict())  # estende le estensioni grazie alla funzione di supporto cover
390
391        return extensions

393    def cover(self, C, t, preset):
394        """
395        Funzione che serve a completare il preset di un evento candidato.
396        Controlla le condizioni concorrenti e sceglie quelle che possono formare un preset
397        per una possibile estensione
398
399        Args:
400
401            C (set()): condizioni concorrenti all'evento e (trovate nella funzione extend_pot_ext())
402        
403            t: transizione candidata a diventare un nuovo evento
404
405            preset (dict()): un dizionario, rappresenta il preset incompleto di t
406        
407        Returns:
408
409            extensions (set()): la possibile estensione
410        """
411        pre_t = self.find_preset(t)
412        extensions = set()
413
414        # se il preset è completo, ovvero se è uguale alla cardinalità di *t entra 
415        # nell'if
416        if len(pre_t) == len(preset): # caso base
417
418            # trova le input_conditions in preset che si trovano nel 
419            # preset di t
420            in_conditions = {preset[p] for p in pre_t}
421
422            # se esiste già un evento uguale nell'unfolding, non si creanno duplicati
423            # ma si utilizza l'evento già inserito
424            existing = self.existing_event(t, in_conditions)
425            if existing:
426                return {existing} # ritorna evento già esistente
427
428            # altrimenti si crea un nuovo evento
429            new_event = Event(t, in_conditions)
430
431            # qui si trovano i posti successivi alla transizione t
432            # sono le output conditions
433            postset_places = {arc.dst for arc in self.net.arcs if arc.src == t}
434
435            for place in postset_places:
436                new_condition = Condition(place, new_event)
437                new_event.output_conditions.add(new_condition)
438
439            # si espandono le estensioni e si ritorna
440            extensions.add(new_event)
441            return extensions
442        else: # passo ricorsivo
443
444            # scelgo un place nel preset NON ancora presente nel preset che sto
445            # costruendo
446            diff = pre_t - set(preset.keys())
447
448            p = next(iter(diff))
449
450            # ciclo le condizioni in C e seleziono quelle che hanno il place uguale a quello scelto precedentemente
451            for d in [cond for cond in C if cond.place == p]:
452                # aggiungo a C_first (C') le condizioni in C concorrenti con la condizione d
453                C_first = {c for c in C if self.is_concurrent(c, d)}
454
455                # incremento il preset in costruzione
456                union_preset = dict(preset)
457                union_preset[p] = d
458
459                # estendo con chiamata ricorsiva
460                extensions |= self.cover(C_first, t, union_preset)
461
462            return extensions

478    def update_conflict_relation(self, event):
479        """
480        Funzione che aggiorna le relazioni di conflitto tra l'evento passato in input e tutti gli eventi presenti
481        nell'unfolding.
482        Controlla ogni evento presente in self.events (che contiene solo gli eventi già presenti nell'unfolding).
483        Propaga il conflitto agli eventi delle input conditions
484
485        Args:
486
487            event (Event): evento a cui si vogliono trovare i conflitti
488        """
489
490        for e in self.events:
491
492            # se "e" è l'evento stesso, continua
493            if e is event:
494                continue
495
496            # se la relazione di conflitto è già presente continua
497            if (event, e) in self.conflict_relation or (e, event) in self.conflict_relation:
498                continue
499
500            # se i preset di event e di "e" non hanno elementi in comune, aggiunge gli eventi nel conflitto
501            if not event.input_conditions.isdisjoint(e.input_conditions):
502                self.add_conflict(event, e)
503            
504            # Se gli eventi delle input conditions di event sono in conflitto con l'evento e, allora anche event
505            # sarà in conflitto con l'evento e
506            for c in event.input_conditions:
507                if c.event and ((c.event, e) in self.conflict_relation or (e, c.event) in self.conflict_relation):
508                    self.add_conflict(event, e)
509                    break

602    def is_causal(self, x, y):
603        """
604        Algoritmo BFS per controllare la causalità di due argomenti passati in input.
605        Se x e y sono condizioni, vengono rese eventi. Se sono condizioni iniziali, allora False.
606        Se sono lo stesso evento -> False
607        Se, ciclando per tutta la "linea" causale si trova l'evento y, allora ritorna True
608
609        Args:
610
611            x (Event or Condition)
612            y (Event or Condition)
613        
614        Returns:
615
616            True or False (Bool)
617        """
618
619        # Ottengo gli eventi di x e y se sono condizioni, altrimenti lascia invariato
620        x_ev = x.event if isinstance(x, Condition) else x
621        y_ev = y.event if isinstance(y, Condition) else y
622
623        # Se uno dei due eventi è nullo, allora non c'è causalità
624        if x_ev is None or y_ev is None:
625            return False
626        
627        # se gli eventi sono gli stessi PRIMA del ciclo, allora non sono causali
628        if x_ev is y_ev:
629            return False
630        
631        # inizializza un set di nodi visitati e una lista che parte dall'evento x
632        visited = set()
633        cone = [x_ev]
634
635        # estrae un nodo da cone, se node == y allora vuol dire che si è trovato y tra i successori
636        # e quindi c'è causalità
637        while cone:
638            node = cone.pop() # prende il primo elemento da cone e lo assegna a node
639
640            if node == y_ev: # nodo y trovato nei successori
641                return True
642            
643            # se il nodo è visitato continua
644            # aggiunge node a visited e crea il set di successori, che andranno ad estendere il cono
645            if node in visited:
646                continue
647            visited.add(node)
648
649            # condizioni prodotte dal nodo preso in considerazione
650            produced_conditions = {cond for (e, cond) in self.causal_relation_post if e is node}
651
652            # eventi successivi alle condizioni prodotte
653            successor_events = {ev2 for (cond2, ev2) in self.causal_relation_pre if cond2 in produced_conditions}
654
655            # estende il cono con i successor events
656            cone.extend(successor_events)
657        
658        # se y non è trovato ritorna falso
659        return False

681    def is_concurrent(self, n1, n2):
682        """
683        Funzione che controlla la concorrenza tra due eventi n1 e n2 passati input. La concorrenza è calcolata
684        "al volo", controllando la presenza di causalità e conflitti.
685
686        Args:
687
688            n1 (Event or Condition)
689            n2 (Evento or Condition)
690        """
691
692        # se n1 è n2 allora non sono concorrenti
693        if n1 is n2:
694            return False
695        
696        # ottengo gli eventi dei nodi, se sono condizioni
697        e1 = self.node_to_event(n1)
698        e2 = self.node_to_event(n2)
699
700        if e1 is not None and e2 is not None:
701            # se e1 ed e2 sono in relazione di causalità allora non sono concorrenti -> False
702            if self.is_causal(e1, e2) or self.is_causal(e2, e1):
703                return False
704            # se e1 ed e2 sono in conflitto -> False
705            if (e1, e2) in self.conflict_relation or (e2, e1) in self.conflict_relation:
706                return False
707            
708            # se gli eventi sono uguali -> false (se provo a fare e1 is e2 non funziona, devo capire perché)
709            if isinstance(n1, Condition) and isinstance(n2, Condition) and n1.event is n2.event:
710                return False
711
712            return True # se passo tutti i check, allora sono concorrenti
713        
714        # se n1 e n2 sono condizioni e sono le stesse allora torna False
715        # possibilmente ridondante, anche perché in teoria non si raggiunge mai questa parte del codice
716        # da valutare se eliminare completamente questo blocco di codice
717        if isinstance(n1, Condition) and isinstance(n2, Condition):
718            if n1 is n2:
719                return False
720            return True
721        
722        return True

824    def export_unfolding_to_json(self, unf, filename="branching_process.json"):
825        """
826        Funzione che esporta l'unfolding in un file json, si basa sui file utilizzati dal sito
827        i love Petri nets. E' un super prototipo, utile per vedere "al volo" se l'unfolidng è corretto, ma 
828        sono chiare solamente gli eventi, dato che nel sito le condizioni non hanno il nome del place che
829        rappresentano, ma si basano su un indice. Dato che il mio unfolding non è ordinato, l'indice dei place
830        è un po' a casaccio.
831        """
832        data = {
833            "places": [],
834            "transitions": [],
835            "arcs": {},
836            "marking": {},
837            "labels": {},
838            "actions": [],
839            "layout": {}
840        }
841
842        place_map = {}
843        trans_map = {}
844        place_idx = 0
845        trans_idx = 0
846
847        for n in unf:
848            # se il nodo è una condition riempio ogni campo del json con i dati di tale condition
849            # alla fine aumento index di 1
850            if isinstance(n, Condition):
851                pname = f"p{place_idx}"
852                place_map[n] = pname
853                data["places"].append(pname)
854                data["labels"][pname] = n.place.name
855                if n.event is None:
856                    data["marking"][pname] = 1
857                place_idx += 1
858            
859            # faccio la stessa cosa nel caso il nodo sia un evento
860            elif isinstance(n, Event):
861                tname = f"t{trans_idx}"
862                trans_map[n] = tname
863                data["transitions"].append(tname)
864                data["labels"][tname] = n.transition.name
865                data["actions"].append(n.transition.name)
866                trans_idx += 1
867
868        # questo blocco di codice fa in modo di non ripetere i nomi delle transizioni duplicate
869        seen = set()
870        actions_unique = []
871        for a in data["actions"]:
872            if a not in seen:
873                seen.add(a)
874                actions_unique.append(a)
875        data["actions"] = actions_unique
876        
877        # crea gli archi tra condizioni ed eventi (o il contrario) in base alla loro relazione di
878        # causalità
879        for (cond, ev) in self.causal_relation_pre:
880            src = place_map.get(cond)
881            dst = trans_map.get(ev)
882            if src and dst:
883                data["arcs"][f"{src},{dst}"] = 1
884
885        for (ev, cond) in self.causal_relation_post:
886            src = trans_map.get(ev)
887            dst = place_map.get(cond)
888            if src and dst:
889                data["arcs"][f"{src},{dst}"] = 1
890
891        # aggiunge tutti i nodi in unfolding
892        nodes_in_unf = set(n for n in unf)
893
894        # Costruzione del grafo che rappresenta l'ordine causale
895        # e calcolo dei gradi per ogni nodo
896        graph = defaultdict(list)
897        indegree = defaultdict(int)
898
899        # aumenta il livello di ogni nodo in base alla relazione causale
900        # se si è in un caso cond -> ev, allora alla condizione si "attacca" l'evento successivo
901        for (cond, ev) in self.causal_relation_pre:
902            if cond in nodes_in_unf and ev in nodes_in_unf:
903                graph[cond].append(ev)
904                indegree[ev] += 1
905                indegree.setdefault(cond, 0)
906        
907        # se si è in un caso ev -> cond, all'evento si attacca la condizione successiva
908        for (ev, cond) in self.causal_relation_post:
909            if ev in nodes_in_unf and cond in nodes_in_unf:
910                graph[ev].append(cond)
911                indegree[cond] += 1
912                indegree.setdefault(ev, 0)
913        
914        for n in nodes_in_unf:
915            indegree.setdefault(n, 0)
916
917        level = {} # distanza dalla marcatura iniziale
918
919        # Si inizializza la coda con i nodi senza predecessori, ovvero quelli al livello
920        # zero
921        queue = deque([n for n in unf if indegree[n] == 0])
922        for n in queue:
923            level[n] = 0
924
925        # visita topologica
926        # ogni nodo successore ottiene level = level predecessore + 1
927        while queue:
928            node = queue.popleft()
929            for succ in graph[node]:
930                level[succ] = max(level.get(succ, 0), level[node] + 1)
931                indegree[succ] -= 1
932
933                # quando arriva allo 0, tutti i predecessori sono stati elaborati
934                if indegree[succ] == 0:
935                    queue.append(succ)
936
937        # i nodi sono stratificati in base al loro livello
938        layers = defaultdict(list)
939        for n, l in level.items():
940
941            if isinstance(n, Condition) and n not in place_map:
942                continue
943            if isinstance(n, Event) and n not in trans_map:
944                continue
945            layers[l].append(n)
946
947        # distanze standard
948        spacing_x = 300
949        spacing_y = 100
950
951        # ad ogni nodo viene assegnata una posizione nel layout
952        # i nodi nello stesso livello sono verticali, i livelli successivi avanzano verticalmente
953        for l, nodes in sorted(layers.items()):
954            for i, n in enumerate(nodes):
955                x = spacing_x * l
956                y = spacing_y * i
957                if isinstance(n, Condition):
958                    name = place_map[n]
959                else:
960                    name = trans_map[n]
961                data["layout"][name] = {"x": x, "y": y}

634

636        # e quindi c'è causalità
637        while cone:
638            node = cone.pop() # prende il primo elemento da cone e lo assegna a node
639

641                return True
642

644            # aggiunge node a visited e crea il set di successori, che andranno ad estendere il cono
645            if node in visited:
646                continue
647            visited.add(node)
648
649            # condizioni prodotte dal nodo preso in considerazione
650            produced_conditions = {cond for (e, cond) in self.causal_relation_post if e is node}
651
652            # eventi successivi alle condizioni prodotte
653            successor_events = {ev2 for (cond2, ev2) in self.causal_relation_pre if cond2 in produced_conditions}
654
655            # estende il cono con i successor events
656            cone.extend(successor_events)
657        
658        # se y non è trovato ritorna falso
659        return False

685

687

693        if n1 is n2:
